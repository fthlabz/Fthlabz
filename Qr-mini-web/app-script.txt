function doGet(e) {
  var action = e.parameter.action;
  
  // 1. VERİ OKUMA
  if (action == "read") {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Sayfa1");
    var data = sheet.getDataRange().getValues();
    var id = e.parameter.id;
    for (var i = 1; i < data.length; i++) {
      if (data[i][0] == id) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "found",
          title: data[i][1], not: data[i][2], folder: data[i][3], ses: data[i][4], pass: data[i][5]
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }
    return ContentService.createTextOutput(JSON.stringify({status: "empty"})).setMimeType(ContentService.MimeType.JSON);
  }
  
  // 2. KAYDETME
  else if (action == "save") {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Sayfa1");
    var id = e.parameter.id;
    var rowData = [id, e.parameter.title, e.parameter.not, e.parameter.folder, e.parameter.ses, "encrypted"];
    var data = sheet.getDataRange().getValues();
    var found = false;
    for (var i = 1; i < data.length; i++) {
      if (data[i][0] == id) { sheet.getRange(i + 1, 1, 1, 6).setValues([rowData]); found = true; break; }
    }
    if (!found) { sheet.appendRow(rowData); }
    return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
  }

  // 3. DOSYALARI ÇEKME (HATA AYIKLAMALI)
  else if (action == "getFiles") {
    var rawLink = e.parameter.url;
    var folderId = "";

    try {
      // LİNK TEMİZLEME: Link ne kadar karışık olursa olsun ID'yi bulur.
      var match = rawLink.match(/[-\w]{25,}/); 
      if (match && match.length > 0) {
        folderId = match[0];
      } else {
        throw "Linkin içinde ID bulunamadı.";
      }

      var folder = DriveApp.getFolderById(folderId);
      var files = folder.getFiles();
      var fileList = [];
      
      while (files.hasNext()) {
        var file = files.next();
        var mime = file.getMimeType();
        
        if (mime.indexOf('image') > -1 || mime.indexOf('video') > -1) {
          fileList.push({
            id: file.getId(),
            type: mime.indexOf('image') > -1 ? 'image' : 'video',
            mime: mime
          });
        }
      }
      return ContentService.createTextOutput(JSON.stringify({status: "success", files: fileList})).setMimeType(ContentService.MimeType.JSON);
      
    } catch (err) {
      return ContentService.createTextOutput(JSON.stringify({
        status: "error", 
        msg: err.toString(), 
        triedId: folderId // Hatayı görmek için
      })).setMimeType(ContentService.MimeType.JSON);
    }
  }
}
